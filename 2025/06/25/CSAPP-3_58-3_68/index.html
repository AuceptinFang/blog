题目 *3.58

一个函数的原型为：

long decode2(long x, long y, long z);

GCC 产生如下汇编代码：

decode2:
    subq    %rdx, %rsi
    imulq   %rsi, %rdi
    movq    %rsi, %rax
    salq    $63, %rax
    sarq    $63, %rax
    xorq    %rdi, %rax
    ret

参数 x、y 和 z 通过寄存器 %rdi、%rsi 和 %rdx 传递。代码将返回值存放在寄存器 %rax 中。写出等价于上述汇编代码的 decode2 的 C 代码。

不算复杂的题，逐行翻译即可，不过由于我的设备上long是4字节的,无法完成移位63的操作，需要把原型换成long long

long decode(long long x, long long y ,long long z){
    y -= z;
    y*=x;
    y = y<<63;
    y = y>>63;
    return x or y;
}

汇编后发现没有movq %rsi %rax的一步，补了一个中间变量后没变化，似乎被优化掉了，那就干脆保持原样吧

subq	%rdx, %rsi
imulq	%rdi, %rsi
salq	$63, %rsi
sarq	$63, %rsi
orq	%rdi, %rsi
setne	%al
movzbl	%al, %eax
ret

对y变量只保留了最后一位，然后异或，大概是一种掩码操作

题目 **3.59

下面的代码计算两个 64 位有符号值的 128 位乘积，并将结果存在内存中：

typedef __int128 int128_t;

void store_prod(int128_t *dest, int64_t x, int64_t y) { 
    *dest = x * (int128_t) y;
}

GCC 产生下面的汇编代码来实现计算：

store_prod:
    movq    %rdx, %rax
    cqto
    movq    %rsi, %rcx
    sarq    $63, %rcx
    imulq   %rax, %rcx
    imulq   %rsi, %rdx
    addq    %rdx, %rcx
    mulq    %rsi
    addq    %rcx, %rdx
    movq    %rax, (%rdi)
    movq    %rdx, 8(%rdi)
    ret

为了满足在 64 位机器上实现 128 位运算所需的多精度计算，这段代码用了三个乘法。描述用来计算乘积的算法，对汇编代码加注释，说明它是如何实现你的算法的。

提示： 在把参数 x 和 y 扩展到 128 位时，它们可以重写为 $x = 2^{64} \cdot x_h + x_l$ 和 $y = 2^{64} \cdot y_h + y_l$，这里 $x_h$、$x_l$、$y_h$ 和 $y_l$ 都是 64 位值。类似地，128 位的乘积可以写成 $p = 2^{64} \cdot p_h + p_l$，这里 $p_h$ 和 $p_l$ 是 64 位值。请解释这段代码是如何用 $x_h$、$x_l$、$y_h$ 和 $y_l$ 来计算 $p_h$ 和 $p_l$ 的。

x和y都是有符号的，所以拓展到128位就是把符号位算数左移64位，其余不变。
默认的128位处理是把低64位放rax，高64位放rdx，但是遇到了负数补码就乱了

$$p = 2^{128} \cdot x_h y_h + 2^{64} \cdot (x_h y_l + x_l y_h) + x_l y_l$$

其中 $2^{128} \cdot x_h y_h$ 直接忽略

store_prod:
    movq    %rdx, %rax    ; rax = y 
    cqto                  ; 将rax符号扩展到rdx，rdx = y_h
    movq    %rsi, %rcx    ; rcx = x 
    sarq    $63, %rcx     ; 算术右移63位提取符号，rcx = x_h
    imulq   %rax, %rcx    ; rcx = y * x_h (即 y_l * x_h)
    imulq   %rsi, %rdx    ; rdx = x * y_h (即 x_l * y_h)
    addq    %rdx, %rcx    ; rcx = y_l * x_h + x_l * y_h
    mulq    %rsi          ; rax:rdx = y_l * x_l (无符号乘法)
    addq    %rcx, %rdx    ; rdx += rcx，得到最终的高64位
    movq    %rax, (%rdi)  ; 存储低64位到dest[0]
    movq    %rdx, 8(%rdi) ; 存储高64位到dest[1]
    ret

总结一下重点，cqto(Convert Quadword to Octoword)指令，直接完成四字到八字的转换，高位放在rdx里
如果要自己完成，移位要用算数右移，因为负数存的是补码

  movq    %rax, %rdi          ;rdi = 3n
  testq   %rax, %rax          ;按位与rax，设置符号位

  salq    $3, %r8             ;r8 *= 2^3 ; r8 = 8(4n + 1)
  leaq    (%rsi,%rdx,8), %rcx ;rcx = rsi + 8 * rdx 
  movl    $0, %eax;           ;eax = 0
  movl    $0, %edx;           ;edx = 0
.L3:
  addq    (%rcx), %rax        ;rax = (rcx)
  addq    $1, %rdx            ;rdx += 1
  addq    %r8, %rcx           ;rcx += r8
  cmpq    %rdi, %rdx          ;
  jne     .L3                 如果rdi和rdx不相等就回到L3
  req; ret
.L4:

  ret                         ;return 0

#include <stdio.h>

#define NR(n) (3 * (n))
#define NC(n) (4 * (n) + 1)

long sum_col(long n, long A[NR(n)][NC(n)], long j) {
    long i;
    long result = 0;
    for (i = 0; i < NR(n); i++) {
        result += A[i][j];
    }
    return result;
}

题目**3.67

这个作业要查看GCC为参数和返回值中有结构的函数产生的代码，由此可看到这些语言特性通常是如何实现的。
下面的C代码中有一个函数process，它用结构作为参数和返回值，还有一个函数eval，它调用process：

typedef struct{
  long a[2];
  long *p;
}strA;

typedef struct{
  long u[2];
  long q;
}strB;

strB process(strA s){
  strB r;
  r.u[0] = s.a[1];
  r.u[1] = s.a[0];
  r.q = *s.p;
  retrun r;
}

long eval(long x, long y, long z){
  strA s;
  s.a[0] = x;
  s.a[1] = y;
  s.p = &z;
  strB r= process(s);
  retrun r.u[0] + r.u[1] + r.q;
}

GCC为这两个函数产生下面的代码：

process: 
  movq    %rdi, %rax          ;rax = rdi
  movq    24(%rsp), %rdx      ;rdx = &z
  movq    (%rdx), %rdx        ;rdx = *(&z) = q
  movq    16(%rsp), %rcx      ;rcx = s.a[1]
  movq    %rcx, (%rdi)        ;*rdi = rcx = s.a[1] 
  movq    8(%rsp), %rcx       ;rcx = s.a[0]
  movq    %rcx, 8(%rdi)       ;*rdi + 8 = s.a[0]
  movq    %rdx, 16(%rdi)      ;*rdi + 16 = rdx = q
  ret

eval:
  subq    $104, %rsp          ;rsp回退13*8，开始存参数
  movq    %rdx, 24(%rsp)      ;s.p = z
  leaq    24(%rsp), %rax      ;rax = &z
  movq    %rdi, (%rsp)        ;s.a[0] = x
  movq    %rsi, 8(%rsp)       ;s.a[1] = y
  movq    %rax, 16(%rsp)      ;s.p = &z
  leaq    64(%rsp), %rdi      ;首地址存在rsp + 64

  movq    72(%rsp), %rax      ;局部变量
  addq    64(%rsp), %rax      
  addq    80(%rsp), %rax      ;r.u[0] + r.u[1] + r.q
  addq    $104, %rsp          ;恢复栈指针
  ret

开猜，B = 8，A = 8 ————8和32都碰上了，但是184没有

尝试2，23时意外碰上了32这个偏移，仔细一看原来是对齐，提醒我们此题的数字是因为要求的四字节对齐产生的

$
5 \leq B \leq 8，

$
